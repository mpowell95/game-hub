import pkg from '/tmp/claude-0/-home-user-game-hub/8b63e124-b89f-544b-807c-895261976e7f/scratchpad/node_modules/playwright-core/index.js';
const { chromium } = pkg;
const PORT = process.argv[2] || 8123;
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome',
  args: ['--use-gl=angle','--use-angle=swiftshader','--enable-unsafe-swiftshader','--no-sandbox'] });
const ctx = await browser.newContext({ viewport: { width: 393, height: 852 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true });
const page = await ctx.newPage();
const warn=[]; page.on('console', m => { const t=m.text(); if(/WebGL|context/i.test(t)) warn.push(m.type()+': '+t.slice(0,140)); });
page.on('pageerror', e => console.log('PAGEERROR', e.message));
await page.addInitScript(() => {
  localStorage.setItem('gamehub.profile', JSON.stringify({ name: 'Perf', emoji: '🐙', opponents: [] }));
  localStorage.setItem('gamehub.skeeball.v1', JSON.stringify({ board: 'brickcity' }));
  localStorage.setItem('gamehub.adminConfig.v1', JSON.stringify({ v: { skeeball: { boards: { brickcity:{open:true,testing:false} } } }, at: Date.now() }));
  // Count every WebGL context this page ever creates, and how many are still alive.
  window.__ctx = [];
  const orig = HTMLCanvasElement.prototype.getContext;
  HTMLCanvasElement.prototype.getContext = function (type, ...rest) {
    const g = orig.call(this, type, ...rest);
    if (g && /webgl/i.test(type)) window.__ctx.push({ gl: g, canvas: this });
    return g;
  };
});
await page.goto(`http://localhost:${PORT}/skeeball/`, { waitUntil: 'domcontentloaded' });
await page.waitForTimeout(1500);
const report = async (label) => {
  const r = await page.evaluate(() => {
    const all = window.__ctx;
    let alive = 0, lost = 0, inDom = 0;
    for (const c of all) { if (c.gl.isContextLost()) lost++; else alive++; if (document.contains(c.canvas)) inDom++; }
    return { created: all.length, alive, lost, inDom };
  });
  console.log(`${label.padEnd(26)} contexts created ${r.created}  still ALIVE ${r.alive}  lost ${r.lost}  canvas still in the DOM ${r.inDom}`);
};
await report('on the gallery');
await (await page.$('[data-role="play"]')).click({ force: true });
await page.waitForSelector('.sk-canvas', { timeout: 10000 }); await page.waitForTimeout(600);
await report('rack 1');
for (let i = 2; i <= 6; i++) {
  await page.click('[data-role="machines"]'); await page.waitForTimeout(300);
  await page.click('[data-role="new"]'); await page.waitForTimeout(700);
  await report(`rack ${i}`);
}
await page.click('[data-role="machines"]'); await page.waitForTimeout(300);
await page.click('[data-role="gallery"]'); await page.waitForTimeout(900);
await report('back on the gallery');
if (warn.length) console.log('\nconsole:', warn.slice(0,6).join('\n         '));
await browser.close();

import pkg from '/tmp/claude-0/-home-user-game-hub/8b63e124-b89f-544b-807c-895261976e7f/scratchpad/node_modules/playwright-core/index.js';
const { chromium } = pkg;
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome',
  args: ['--use-gl=angle','--use-angle=swiftshader','--enable-unsafe-swiftshader','--no-sandbox'] });
const ctx = await browser.newContext({ viewport: { width: 393, height: 852 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true });
const page = await ctx.newPage();
page.on('pageerror', e => console.log('PAGEERROR', e.message));
await page.addInitScript(() => {
  localStorage.setItem('gamehub.profile', JSON.stringify({ name: 'Perf', emoji: '🐙', opponents: [] }));
  localStorage.setItem('gamehub.skeeball.v1', JSON.stringify({ board: 'brickcity' }));
  localStorage.setItem('gamehub.adminConfig.v1', JSON.stringify({ v: { skeeball: { boards: { brickcity:{open:true,testing:false} } } }, at: Date.now() }));
});
await page.goto('http://localhost:8123/skeeball/', { waitUntil: 'domcontentloaded' });
await page.waitForTimeout(1500);
await (await page.$('[data-role="play"]')).click({ force: true });
await page.waitForSelector('.sk-canvas', { timeout: 10000 });
await page.waitForTimeout(600);

await page.evaluate(() => {
  const ui = window.__skTest.ui, R = ui.renderer;
  window.__T = [];                       // per-throw frame logs
  window.__cur = null;
  const origR = R.render.bind(R);
  R.render = (game, dt) => {
    origR(game, dt);
    if (!window.__cur) return;
    const live = (game && game.balls) || [];
    const drawn = R._balls.filter(m => m.visible)
      .map(m => ({ x:+m.position.x.toFixed(4), y:+m.position.y.toFixed(4), z:+m.position.z.toFixed(4) }))
      .filter(m => !(Math.abs(m.x) < 1e-6 && Math.abs(m.z + 0.12) < 1e-6));
    const phys = live.map(s => { const p=s.ball.position, v=s.ball.velocity;
      return { x:+p.x.toFixed(4), y:+p.y.toFixed(4), z:+p.z.toFixed(4), sp:+Math.hypot(v.x,v.y,v.z).toFixed(3), cap: s.captured || null }; });
    window.__cur.f.push({ t: performance.now(), nLive: live.length, drawn, phys });
  };
});

// (power, aim) pairs the headless sweep says land in the -20 / -10 / -20 row
const SHOTS = [[0.30,0.0],[0.325,0.0],[0.35,0.0],[0.30,-0.3],[0.325,0.3],[0.35,-0.2],
               [0.375,0.0],[0.30,0.5],[0.325,-0.5],[0.35,0.4],[0.275,0.0],[0.40,0.1]];
const results = [];
for (const [power, aim] of SHOTS) {
  const ok = await page.evaluate(([power, aim]) => {
    const ui = window.__skTest.ui;
    if (!ui.game || !ui.game.canThrow()) return false;
    window.__cur = { power, aim, f: [] };
    return ui.game.throwBall({ power, aim });
  }, [power, aim]);
  if (!ok) { await page.waitForTimeout(1200); continue; }
  await page.waitForFunction(() => window.__skTest.ui.game && window.__skTest.ui.game.balls.length === 0, { timeout: 20000 }).catch(()=>{});
  await page.waitForTimeout(1200);
  const r = await page.evaluate(() => {
    const c = window.__cur; window.__cur = null;
    const f = c.f;
    if (!f.length) return { power: c.power, aim: c.aim, note: 'no frames' };
    // longest stretch where the DRAWN ball did not move
    let run = 0, best = 0, at = null, last = '';
    for (const fr of f) { const k = JSON.stringify(fr.drawn);
      if (fr.drawn.length && k === last) { run++; if (run > best) { best = run; at = { pos: fr.drawn[0], nLive: fr.nLive, phys: fr.phys[0] || null }; } }
      else run = 0;
      last = k; }
    const dur = (f[f.length-1].t - f[0].t) / 1000;
    const capFrames = f.filter(x => x.phys.some(p => p.cap)).length;
    const slowFrames = f.filter(x => x.phys.some(p => p.sp < 0.12)).length;
    return { power: c.power, aim: c.aim, frames: f.length, seconds: +dur.toFixed(2),
      capturedFrames: capFrames, nearStillFrames: slowFrames,
      longestFrozenDrawn: best, frozenAt: at };
  });
  results.push(r);
  console.log(`p${String(r.power).padEnd(5)} a${String(r.aim).padEnd(5)} ${String(r.seconds).padStart(5)}s  frames ${String(r.frames).padStart(4)}  captured ${String(r.capturedFrames).padStart(3)}  near-still ${String(r.nearStillFrames).padStart(3)}  longest FROZEN drawn ${r.longestFrozenDrawn}`);
  if (r.frozenAt && r.longestFrozenDrawn > 8) console.log('     frozen at', JSON.stringify(r.frozenAt));
}
const sk = await page.evaluate(() => { try { return ((JSON.parse(localStorage.getItem('gamehub.stats')||'{}').games||{}).skeeball||{}).sk||{}; } catch { return {}; } });
console.log('\nscore this rack:', await page.evaluate(() => window.__skTest.ui.game ? window.__skTest.ui.game.score : '(rack over)'));
await browser.close();

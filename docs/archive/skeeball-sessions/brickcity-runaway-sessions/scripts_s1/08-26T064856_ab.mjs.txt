    // Count REAL canvas frames: hash the canvas each animation frame for 3s and count changes.
    return page.evaluate(() => new Promise(res => {
      const c = document.querySelector('.sk-canvas');
      const t = document.createElement('canvas'); t.width = 40; t.height = 40;
      const x = t.getContext('2d');
      let last = '', changes = 0, ticks = 0;
      const t0 = performance.now();
      const tick = () => {
        ticks++;
        x.drawImage(c, 0, 0, c.width, c.height, 0, 0, 40, 40);
        const d = t.getContext('2d').getImageData(0,0,40,40).data;
        let h = 0; for (let i = 0; i < d.length; i += 17) h = (h * 31 + d[i]) | 0;
        const s = String(h); if (s !== last) { changes++; last = s; }
        if (performance.now() - t0 < 3000) requestAnimationFrame(tick);
        else res({ ticks, changes, secs: (performance.now()-t0)/1000 });
      };
      requestAnimationFrame(tick);
    }));
  };

    for(let i=1;i<=steps;i++){await cdp.send('Input.dispatchTouchEvent',{type:'touchMove',touchPoints:[{x,y:y0-dist*i/steps,id:1}]});await page.waitForTimeout(ms/steps);}

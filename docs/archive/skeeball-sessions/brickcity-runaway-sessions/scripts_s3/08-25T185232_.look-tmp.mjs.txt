import { chromium } from 'playwright-core';
const OUT = '/tmp/claude-0/-home-user-game-hub/0d02ea3d-1ee6-5bae-b77e-b8fedfb9ebcd/scratchpad';
const b = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome' });
const run = async (w, h, tag) => {
  const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 2, hasTouch: true, isMobile: true });
  const page = await ctx.newPage();
  page.on('pageerror', (e) => console.log('PAGE ERROR:', e.message));
  await page.addInitScript(() => {
    localStorage.setItem('gamehub.profile', JSON.stringify({ name: 'Look', emoji: '🦖', color: '#1F5FA8' }));
    localStorage.setItem('gamehub.skeeball.v1', JSON.stringify({ board: 'brickcity' }));
    localStorage.setItem('gamehub.adminConfig.v1', JSON.stringify({ games: {}, skeeball: { boards: { brickcity: { open: true, testing: false } } }, corrections: { skeeball: {} } }));
  });
  await page.goto('http://localhost:8123/skeeball/', { waitUntil: 'domcontentloaded' });
  await page.waitForSelector('.sk-slide', { timeout: 30000 });
  await page.waitForTimeout(1200);
  await page.locator('button:has-text("Play")').first().click();
  await page.waitForSelector('.sk-goal', { timeout: 45000 });
  await page.waitForTimeout(2500);
  await page.screenshot({ path: `${OUT}/${tag}-play.png` });
  const geo = await page.evaluate(() => {
    const r = (sel) => { const el = document.querySelector(sel); if (!el) return null; const q = el.getBoundingClientRect(); return { t: Math.round(q.top), b: Math.round(q.bottom), w: Math.round(q.width), h: Math.round(q.height) }; };
    const boxes = [...document.querySelectorAll('.sk-goal')].map((el) => { const q = el.getBoundingClientRect(); return { label: el.querySelector('em')?.textContent, val: el.querySelector('b')?.textContent, t: Math.round(q.top), b: Math.round(q.bottom), w: Math.round(q.width), h: Math.round(q.height) }; });
    const g = document.querySelector('.sk-gtotal');
    const gs = g && getComputedStyle(g, '::after');
    return { boxes, gtotal: r('.sk-gtotal'), gtotalHit: gs && gs.height, swipe: r('.sk-swipe'),
      sideScroll: document.scrollingElement.scrollWidth > window.innerWidth };
  });
  console.log(`\n=== ${tag} (${w}x${h}) ===`);
  console.log(JSON.stringify(geo));
  const gap = geo.swipe.t - Math.max(...geo.boxes.map((x) => x.b));
  console.log(`rail bottom to swipe top: ${gap}px`);
  const hits = await page.evaluate(() => {
    const at = (x, y) => { const el = document.elementFromPoint(x, y); return el ? String(el.className || el.tagName).slice(0, 24) : 'none'; };
    const out = {};
    document.querySelectorAll('[data-def]').forEach((el) => { const q = el.getBoundingClientRect(); out[el.getAttribute('data-def')] = at(q.left + q.width / 2, q.top + q.height / 2); });
    const sw = document.querySelector('.sk-swipe').getBoundingClientRect();
    out['swipe-top'] = at(sw.left + sw.width / 2, sw.top + 3);
    out['swipe-mid'] = at(sw.left + sw.width / 2, sw.top + sw.height / 2);
    return out;
  });
  console.log('hit test:', JSON.stringify(hits));
  await page.locator('.sk-goal').first().tap();
  await page.waitForSelector('.sk-def-modal', { timeout: 8000 });
  await page.waitForTimeout(400);
  await page.screenshot({ path: `${OUT}/${tag}-sheet.png` });
  console.log('sheet:', JSON.stringify(await page.evaluate(() => {
    const m = document.querySelector('.sk-def-modal');
    const x = m.querySelector('.sk-def-x').getBoundingClientRect();
    return { h: Math.round(m.getBoundingClientRect().height),
      x: { w: Math.round(x.width), h: Math.round(x.height) },
      rows: [...m.querySelectorAll('.sk-def-row')].map((r) => ({ l: r.querySelector('em').textContent, p: r.querySelector('b').textContent, d: r.querySelector('.sk-def-text').textContent, focus: r.classList.contains('is-focus') })) };
  }), null, 1));
  await page.locator('.sk-def-x').tap();
  await page.waitForTimeout(300);
  console.log('closes:', await page.evaluate(() => !document.querySelector('.sk-def-modal')));
  // and from the wide total bar
  await page.locator('.sk-gtotal').tap();
  await page.waitForTimeout(400);
  console.log('total bar opens it too:', await page.evaluate(() => !!document.querySelector('.sk-def-modal')));
  await page.screenshot({ path: `${OUT}/${tag}-sheet-net.png` });
  await ctx.close();
};
await run(375, 667, 'small');
await run(393, 852, 'tall');
await b.close();
console.log('done');

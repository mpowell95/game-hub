import { firebaseConfig } from '/home/user/game-hub/js/firebase-config.js';
import { signInAnonymously, readPath } from '/home/user/game-hub/backups/rtdb-backup.mjs';
const tok = await signInAnonymously(firebaseConfig.apiKey);
const players = await readPath(firebaseConfig.databaseURL, 'players', tok);
const tally = {}; let withSk = 0, totalPlayers = 0;
const byPerson = {};
for (const id of Object.keys(players || {})) {
  totalPlayers++;
  const p = players[id] || {};
  const sk = ((p.stats || {}).games || {}).skeeball;
  if (!sk) continue;
  withSk++;
  const nm = (p.profile || {}).name || id.slice(0,8);
  for (const [bid, b] of Object.entries(sk.boards || {})) {
    tally[bid] = (tally[bid] || 0) + (b.plays | 0);
    byPerson[nm] = byPerson[nm] || {};
    byPerson[nm][bid] = (byPerson[nm][bid] || 0) + (b.plays|0);
  }
  if (sk.practice) tally['(practice bucket)'] = (tally['(practice bucket)'] || 0) + 1;
}
console.log('player nodes', totalPlayers, '| with skeeball stats', withSk);
console.log('plays per board:', tally);
console.log('per person:', JSON.stringify(byPerson, null, 1).slice(0, 1500));
const thr = await readPath(firebaseConfig.databaseURL, 'skeeballThrows', tok);
const boards = {};
for (const dev of Object.keys(thr||{})) for (const k of Object.keys(thr[dev]||{})) {
  const r = thr[dev][k]; if (r && r.board) boards[r.board] = (boards[r.board]||0)+1;
}
console.log('logged throws per board:', boards);

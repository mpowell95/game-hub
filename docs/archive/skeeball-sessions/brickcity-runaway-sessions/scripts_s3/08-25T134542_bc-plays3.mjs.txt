import { firebaseConfig } from '/home/user/game-hub/js/firebase-config.js';
import { signInAnonymously, readPath } from '/home/user/game-hub/backups/rtdb-backup.mjs';
const tok = await signInAnonymously(firebaseConfig.apiKey);
const players = await readPath(firebaseConfig.databaseURL, 'players', tok);
const tally = {}; const bc = [];
for (const id of Object.keys(players||{})) {
  const p = players[id]||{};
  const g = ((p.stats||{}).games||{}).skeeball; if (!g) continue;
  const sk = g.sk || {};
  const nm = (p.profile||{}).name || id.slice(0,8);
  for (const [bid,b] of Object.entries(sk.boards||{})) tally[bid]=(tally[bid]||0)+(b.plays|0);
  const b = (sk.boards||{}).brickcity;
  if (b) bc.push({nm, plays:b.plays|0, best:b.best|0, pts:b.points|0, clean:b.cleanRacks|0,
    slots:Object.keys(b.slots||{}).filter(k=>b.slots[k]).length, unlocked: !!(sk.unlocked||{}).brickcity});
  if (sk.practice) tally['(practice)'] = (tally['(practice)']||0) + (sk.practice.plays|0 || 1);
}
console.log('plays per board:', tally);
console.log('brickcity records:', bc.length);
for (const r of bc) console.log(' ', r);

async function records(label) {
  const h = await p.$eval('.lb-h3', (e)=>e.textContent).catch(()=>'(no heading)');
  const cells = await p.$$eval('.lb-rec', (els)=>els.map((e)=>({
    v:(e.querySelector('b')||{}).textContent||'', l:(e.querySelector('span')||{}).textContent||'',
    who:(e.querySelector('em')||{}).textContent||'' })));
  const cards = await p.$$eval('.lb-plist .lb-pcard', (els)=>els.map((e)=>
    ((e.querySelector('.lb-pname')||{}).textContent||'').trim()+' · '+((e.querySelector('.lb-pnum b')||{}).textContent||'')));
  console.log('\n=== '+label+' ===');
  console.log('  heading: '+h);
  console.log('  cards:   '+JSON.stringify(cards));
  for (const c of cells) console.log('   %s  %s  (%s)', c.v.padStart(8), c.l.padEnd(16), c.who.trim());
}
await records('ALL MACHINES');
await p.screenshot({path:OUT+'/rec-all.png'});

  const opt = p.locator('.lb-panel-list [data-machine]', { hasText: machine }).first();

}

async function read(label) {
  const rows = await p.$$eval('.lb-plist .lb-pcard', (els) => els.map((el) => ({
    rank: (el.querySelector('.lb-chip') || {}).textContent || '',
    name: (el.querySelector('.lb-pname') || {}).textContent || '',
    sub:  (el.querySelector('.lb-psubline') || {}).textContent || '',
    big:  (el.querySelector('.lb-pnum b') || {}).textContent || '',
    unit: (el.querySelector('.lb-pnum span') || {}).textContent || '',
  })));
  console.log('\n=== ' + label + ' ===');
  console.log('  rank | name             | BIG      unit    | subline');
  for (const r of rows.slice(0, 8)) {
    console.log('  %s | %s | %s %s | %s', r.rank.padStart(4), r.name.trim().padEnd(16),
      r.big.padStart(7), r.unit.trim().padEnd(7), r.sub.trim());
  }
  return rows;
}

const sortBtns = await p.$$eval('[data-sort], .lb-seg, .lb-pill, button', (els) =>

const ranks = played.map((r) => parseInt(r.rank.replace(/[^0-9]/g, ''), 10));
console.log('\n  played values down the list:', nums.slice(0, 8).join(', '));
console.log('  descending?              ', nums.every((v, i) => i === 0 || v <= nums[i - 1]) ? 'YES' : 'NO');
console.log('  rank badges:             ', ranks.slice(0, 8).join(', '));
console.log('  badges in order?         ', ranks.every((v, i) => i === 0 || v >= ranks[i - 1]) ? 'YES' : 'NO');

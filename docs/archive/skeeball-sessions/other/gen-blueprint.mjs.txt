const n1 = (n) => (+n).toFixed(1);
function head(tx, ty, ux, uy, s = 9) {
  const bx = tx - ux * s, by = ty - uy * s, px = -uy, py = ux, w = 5;
  return `<polygon class="ah" points="${n1(tx)},${n1(ty)} ${n1(bx + px * w)},${n1(by + py * w)} ${n1(bx - px * w)},${n1(by - py * w)}"/>`;
}
// letter badge at (bx,by); optional arrow to (tx,ty)
function tag(letter, bx, by, tx, ty) {
  let arw = '';
  if (tx != null) {
    const dx = tx - bx, dy = ty - by, L = Math.hypot(dx, dy) || 1, ux = dx / L, uy = dy / L;
    const sx = bx + ux * 13, sy = by + uy * 13;
    arw = `<line class="arw" x1="${n1(sx)}" y1="${n1(sy)}" x2="${n1(tx)}" y2="${n1(ty)}"/>${head(tx, ty, ux, uy)}`;
  }
  return `${arw}<circle class="bdg" cx="${n1(bx)}" cy="${n1(by)}" r="12"/><text class="bdgt" x="${n1(bx)}" y="${n1(by + 5)}">${letter}</text>`;
}
const holeText = (x, y, id) => `<text class="hid" x="${n1(x)}" y="${n1(y)}">${id}</text>`;

const W3 = 640, H3 = 720;
const sub = (a, b) => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const dot = (a, b) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const cross = (a, b) => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const norm = (a) => { const l = Math.hypot(...a) || 1; return [a[0] / l, a[1] / l, a[2] / l]; };
const camC = [0, 0.60, 1.32];
const tgt = F2W(0, G.boardLen * 0.42, 0);
const fwd = norm(sub([tgt[0], tgt[1], tgt[2]], camC));
const rgt = norm(cross(fwd, [0, 1, 0]));
const upv = cross(rgt, fwd);
const fovy = 42 * Math.PI / 180, ff = 1 / Math.tan(fovy / 2), aspect = W3 / H3;
function proj(p) {
  const d = sub(p, camC);
  const x = dot(d, rgt), y = dot(d, upv), z = dot(d, fwd);
  return [W3 / 2 * (1 + (ff / aspect) * x / z), H3 / 2 * (1 - ff * y / z), z];
}
